{
	"enableProgressBar": false,
	"progressBar": {
		"zoneId_regex": ".*$",
		"moveDelay_color": "white",
		"moveDelay_overlay": "progress",
		"moveTime_color": "white",
		"moveTime_overlay": "progress"
	}
}
```
